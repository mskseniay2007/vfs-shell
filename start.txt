# Проверяем работу VFS

# Смотрим корень
ls

# Идём в /home
cd home
ls

# Идём в /home/user
cd user
ls

# Поднимаемся на уровень вверх
cd ..

# Возвращаемся в корень
cd /

# Смотрим конкретную папку, не переходя в неё
ls /etc/config

# Проверяем обработку ошибок
cd /nonexistent
ls /also/not/exist

# Проверяем команды с относительными путями
cd home/user
ls
cd ../../etc
ls

exit
