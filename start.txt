# Тестовый скрипт для этапа 5

# Смотрим стартовое состояние
ls
cd home
cd user
ls

# Создаём новый файл в текущей папке
touch newfile.txt
ls

# Создаём файл с абсолютным путём
touch /home/user/another.txt
ls

# Повторный touch — не ошибка, просто ничего не делает
touch newfile.txt

# touch в несуществующей папке — ошибка
touch /no/such/dir/file.txt

# Создаём папку в текущей папке
mkdir newdir
ls

# Создаём вложенную папку
mkdir /home/user/subdir
ls

# Повторный mkdir — ошибка
mkdir newdir

# mkdir в несуществующем месте — ошибка
mkdir /no/such/dir/newfolder

# Без аргумента — ошибка
touch
mkdir

# Возвращаемся в корень
cd /
ls

# Проверяем команды из этапа 4
uname
cd home/user
uniq dups.txt
cd /

exit
